cmake_minimum_required(VERSION 3.20)
project(FluxerPP LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(${CMAKE_SOURCE_DIR}/include)

# Fluxer++ library
add_library(fluxerpp
    src/FluxerClient.cpp
    src/RestClient.cpp
    src/GatewayClient.cpp
    src/EventDispatcher.cpp
    src/winhttp_send.cpp
)

target_link_libraries(fluxerpp
    winhttp
    ws2_32
    curl
)

# ping_bot example
add_executable(ping_bot
    examples/ping_bot.cpp
)

set_target_properties(ping_bot PROPERTIES LINK_FLAGS "-Wl,-subsystem,console")

target_link_libraries(ping_bot
    fluxerpp
    winhttp
    ws2_32
    curl
)
