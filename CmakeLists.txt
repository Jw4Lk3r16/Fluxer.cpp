# -----------------------------
# INSTALL RULES (MSYS2 + CMake)
# -----------------------------

# Install library (DLL, import lib, static lib)
install(
    TARGETS fluxerpp
    EXPORT Fluxer++Targets
    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}     # fluxerpp.dll
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}     # libfluxerpp.dll.a
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}     # libfluxerpp.a
)

# Install headers
install(
    DIRECTORY include/fluxerpp/
    DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/fluxerpp
)

# Export CMake targets
install(
    EXPORT Fluxer++Targets
    FILE Fluxer++Targets.cmake
    NAMESPACE Fluxer++::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Fluxer++
)

# Generate version file
include(CMakePackageConfigHelpers)

write_basic_package_version_file(
    "${CMAKE_CURRENT_BINARY_DIR}/Fluxer++ConfigVersion.cmake"
    VERSION ${PROJECT_VERSION}
    COMPATIBILITY SameMajorVersion
)

# Configure main config file
configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Fluxer++Config.cmake.in"
    "${CMAKE_CURRENT_BINARY_DIR}/Fluxer++Config.cmake"
    @ONLY
)

# Install config + version files
install(
    FILES
        "${CMAKE_CURRENT_BINARY_DIR}/Fluxer++Config.cmake"
        "${CMAKE_CURRENT_BINARY_DIR}/Fluxer++ConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Fluxer++
)
