cmake_minimum_required(VERSION 3.16)
project(FluxerPP LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(CURL REQUIRED)

add_library(fluxerpp
    src/FluxerClient.cpp
    src/RestClient.cpp
    src/GatewayClient.cpp
    src/EventDispatcher.cpp
)

target_include_directories(fluxerpp PUBLIC include)
target_link_libraries(fluxerpp PRIVATE ${CURL_LIBRARIES})

add_executable(ping_bot examples/ping_bot.cpp)
target_link_libraries(ping_bot PRIVATE fluxerpp ${CURL_LIBRARIES})
